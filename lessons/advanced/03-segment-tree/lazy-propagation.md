## แก้ทีละหลายช่อง

segment tree ในบทที่แล้วแก้ได้ทีละ **จุดเดียว** ทีนี้ถ้าโจทย์สั่งว่า "บวก $v$ ให้ทุกตัวใน $[l, r]$" ล่ะ? ถ้าเรียก update จุดเดียวซ้ำ $r - l + 1$ ครั้ง คำสั่งเดียวก็ใช้ $O(N \log N)$ แล้ว แย่กว่าอาเรย์ธรรมดาเสียอีก

ลองนึกถึงงานจริงดูก่อนนะ: หัวหน้าสั่งว่า "พนักงานแผนก A ทุกคนได้ขึ้นเงินเดือน 500" เราไม่จำเป็นต้องวิ่งไปแก้สลิปทุกใบทันที แค่จดไว้ที่หน้าแฟ้มแผนก A ว่า "ทุกคนในนี้ +500" แล้วแก้ยอดรวมของแผนกได้เลย ใครมาขอดูสลิปรายคนเมื่อไร ค่อยเปิดแฟ้มแล้วส่งโน้ตนั้นลงไปให้คนที่ต้องดู

นี่คือ **lazy propagation**: เลื่อนงานไว้ก่อน ทำเท่าที่จำเป็นต้องทำตอนนั้น

## ไอเดีย: ป้ายค้าง (lazy tag)

แต่ละโหนดมีของสองอย่าง

* **ค่าของโหนด** (เช่น ค่าน้อยสุดของช่วง) ซึ่ง **ถูกต้องเสมอ** รวมผลของทุกคำสั่งที่มาถึงโหนดนี้แล้ว
* **ป้ายค้าง** `lz[v]` = งานที่โหนดนี้รับมาแล้วแต่ **ยังไม่ได้ส่งต่อให้ลูก** ลูกของมันจึงอาจยังเก็บค่าเก่าอยู่

มีสองการกระทำที่ต้องเขียน

* `apply(v, x)`: สั่งงาน "บวก $x$ ทั้งช่วง" ให้โหนด `v` คือแก้ค่าของโหนดทันที แล้วบวก $x$ เข้าป้ายค้าง
* `push(v)`: ส่งป้ายค้างของ `v` ลงไปให้ลูกทั้งสอง (เรียก `apply` กับลูก) แล้วล้างป้ายของ `v`

แล้วเพิ่มกฎเดียวเข้าไปใน update และ query ของบทที่แล้ว:

1. ถ้าช่วงของโหนด **อยู่ข้างใน** $[l, r]$ ทั้งก้อน: update เรียก `apply` แล้ว **หยุด ไม่ต้องลงไปต่อ**
2. **ก่อนจะลงไปหาลูกเมื่อไร ต้อง `push` ก่อนเสมอ** ทั้งใน update และ query เพราะลูกยังไม่ได้รับงานที่ค้าง

ทำไมค่าของโหนดถึงแก้ได้ทันทีโดยไม่ต้องดูลูก? เพราะถ้าทุกตัวในช่วงบวก $x$ เท่ากันหมด ค่าน้อยสุดของช่วงก็บวก $x$ เหมือนกัน ข้อนี้คือเงื่อนไขของ lazy: **ต้องคำนวณค่าใหม่ของทั้งก้อนได้จากค่าเก่ากับงานที่สั่งอย่างเดียว**

### Template: บวกทั้งช่วง + ค่าน้อยสุดของช่วง

คำสั่ง `1 l r x` คือบวก $x$ ให้ $a_l..a_r$ ($x$ ติดลบได้) และ `2 l r` คือถามค่าน้อยสุดของ $a_l..a_r$

```cpp
#include <bits/stdc++.h>
using namespace std;

int n;
vector<long long> a, mn, lz;   // mn = ค่าน้อยสุดของช่วง, lz = ค่าที่ยังค้างบวกให้ลูก

// บวก x ให้ทั้งช่วงของโหนด v: แก้ค่าของ v ทันที แล้วจดว่าลูกยังค้างอยู่
void apply(int v, long long x) {
  mn[v] += x;      // ทุกตัวในช่วงบวก x  ค่าน้อยสุดก็บวก x
  lz[v] += x;
}

// ก่อนจะลงไปหาลูก ต้องส่งของที่ค้างลงไปก่อน
void push(int v) {
  if (lz[v] != 0) {
    apply(2 * v, lz[v]);
    apply(2 * v + 1, lz[v]);
    lz[v] = 0;
  }
}

void build(int v, int tl, int tr) {
  if (tl == tr) { mn[v] = a[tl]; return; }
  int tm = (tl + tr) / 2;
  build(2 * v, tl, tm);
  build(2 * v + 1, tm + 1, tr);
  mn[v] = min(mn[2 * v], mn[2 * v + 1]);
}

// บวก x ให้ a[l..r]
void update(int v, int tl, int tr, int l, int r, long long x) {
  if (r < tl || tr < l) return;
  if (l <= tl && tr <= r) { apply(v, x); return; }   // คลุมทั้งก้อน: จดไว้แล้วหยุด
  push(v);
  int tm = (tl + tr) / 2;
  update(2 * v, tl, tm, l, r, x);
  update(2 * v + 1, tm + 1, tr, l, r, x);
  mn[v] = min(mn[2 * v], mn[2 * v + 1]);
}

// ค่าน้อยสุดของ a[l..r]
long long query(int v, int tl, int tr, int l, int r) {
  if (r < tl || tr < l) return LLONG_MAX;
  if (l <= tl && tr <= r) return mn[v];
  push(v);                                            // ห้ามลืม: ลูกยังไม่ได้รับของที่ค้าง
  int tm = (tl + tr) / 2;
  return min(query(2 * v, tl, tm, l, r), query(2 * v + 1, tm + 1, tr, l, r));
}

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int q;
  cin >> n >> q;
  a.assign(n + 1, 0);
  mn.assign(4 * n, 0);
  lz.assign(4 * n, 0);
  for (int i = 1; i <= n; i++) cin >> a[i];
  build(1, 1, n);
  while (q--) {
    int type, l, r;
    cin >> type >> l >> r;
    if (type == 1) {                 // 1 l r x : บวก x ให้ a[l..r]
      long long x;
      cin >> x;
      update(1, 1, n, l, r, x);
    } else {                         // 2 l r : ค่าน้อยสุดของ a[l..r]
      cout << query(1, 1, n, l, r) << '\n';
    }
  }
}
```

### ลองไล่ด้วยมือ

`a = [3, 1, 4, 1, 5, 9, 2, 6]` ต้นไม้ของค่าน้อยสุดตอนเริ่มคือ $[1, 2] = 1$, $[3, 4] = 1$, $[5, 6] = 5$, $[7, 8] = 2$, $[1, 4] = 1$, $[5, 8] = 2$ และราก $= 1$

**คำสั่ง `1 1 6 5`** (บวก $5$ ให้ $[1, 6]$): รากทับบางส่วน ลงไปลูก โหนด $[1, 4]$ อยู่ข้างในทั้งก้อน `apply` แล้วหยุด โหนด $[5, 8]$ ทับบางส่วน ลงไปต่อ โหนด $[5, 6]$ อยู่ข้างใน `apply` แล้วหยุด ส่วน $[7, 8]$ ไม่ทับ ขากลับคิด $[5, 8]$ กับรากใหม่

```text
                         [1,8] = 2
               /                          \
     [1,4] = 6   lz=5                  [5,8] = 2
       /           \                   /            \
  [1,2] = 1     [3,4] = 1      [5,6] = 10  lz=5   [7,8] = 2
  (ยังเป็นค่าเก่า) (ยังเป็นค่าเก่า)   /      \           /    \
                                5        9        2      6
```

มีแค่สองโหนดที่ได้ป้าย ข้างล่างของมันยังเป็นค่าเก่าทั้งหมด และไม่เป็นไร เพราะยังไม่มีใครถามลงไปถึงตรงนั้น

**คำสั่ง `2 3 5`** (ค่าน้อยสุดของ $[3, 5]$)

| โหนด | เกิดอะไรขึ้น |
|---|---|
| $[1, 8]$ | ทับบางส่วน `push` (ไม่มีป้าย) แล้วลงไปลูก |
| $[1, 4]$ | ทับบางส่วน **`push`**: $[1, 2]$ กลายเป็น $6$, $[3, 4]$ กลายเป็น $6$ ทั้งคู่ได้ `lz=5` ส่วน $[1, 4]$ ล้างป้าย |
| $[1, 2]$ | ไม่ทับ |
| $[3, 4]$ | อยู่ข้างใน ได้ $6$ |
| $[5, 8]$ | ทับบางส่วน `push` (ไม่มีป้าย) |
| $[5, 6]$ | ทับบางส่วน **`push`**: ใบ $5$ กลายเป็น $10$ ใบ $9$ กลายเป็น $14$ |
| $[5, 5]$ | อยู่ข้างใน ได้ $10$ |
| $[6, 6]$ | ไม่ทับ |

คำตอบ $\min(6, 10) = 6$ ตรงกับอาเรย์จริงที่ตอนนี้เป็น $[8, 6, 9, 6, 10, 14, 2, 6]$ ถ้าลืม `push` ที่ $[1, 4]$ เราจะได้ $[3, 4] = 1$ ซึ่งเป็นค่าเก่า

ลองป้อน `8 5`, `3 1 4 1 5 9 2 6`, `1 1 6 5`, `2 3 5`, `1 4 8 -3`, `2 1 8`, `2 5 6` ให้ template จะได้ `6`, `-1`, `7`

### ความซับซ้อน

update แบบช่วงเดินผ่านโหนด **ชุดเดียวกับ query** ของช่วงเดียวกันเป๊ะ (หยุดที่โหนดที่อยู่ข้างใน, แตกต่อเฉพาะโหนดที่ทับบางส่วน) จึงเป็น $O(\log N)$ ส่วน `push` กับ `apply` เป็น $O(1)$ ต่อโหนด รวมทั้งโปรแกรม $O(N + Q \log N)$ เหมือนบทที่แล้ว แต่ตอนนี้แก้ได้ทั้งช่วง

## เปลี่ยนเป็นผลรวม: ป้ายต้องคูณความยาว

ถ้าโหนดเก็บ **ผลรวม** แทนค่าน้อยสุด "บวก $x$ ทั้งช่วง" ไม่ได้ทำให้ผลรวมเพิ่ม $x$ แต่เพิ่ม $x \times$ (จำนวนตัวในช่วง) ดังนั้น `apply` ต้องรู้ความยาวของช่วงด้วย จะส่ง `tl, tr` เข้าไปหรือเก็บความยาวไว้ในโหนดก็ได้

```cpp
// โหนด v ดูแล [tl, tr]
void apply(int v, int tl, int tr, long long x) {
  sum[v] += x * (tr - tl + 1);   // ทุกตัวบวก x  ผลรวมบวก x คูณจำนวนตัว
  lz[v] += x;                    // ป้ายยังเป็น "ตัวละ x" ไม่ได้คูณ
}
```

ป้ายเก็บ "งานที่ค้างต่อหนึ่งตัว" ส่วนค่าของโหนดเก็บ "ผลของงานนั้นต่อทั้งก้อน" สองอย่างนี้คนละหน่วยกัน แยกให้ออก แล้ว `push` ก็ต้องส่งช่วงของลูกเข้าไปด้วย

:::problem lp_lazy01_coin_towers

:::hint ใบ้ Coin Towers
เปลี่ยน template เป็นผลรวม ตามหัวข้อข้างบน identity ของผลรวมคือ $0$ ระวังเลขใหญ่: กองหนึ่งได้ถึงประมาณ $5 \times 10^{13}$ เหรียญ และผลรวมทั้งแถวเกือบ $2.6 \times 10^{18}$ ทุกตัวแปรที่เกี่ยวกับผลรวมและป้ายต้องเป็น `long long` รวมถึงตอนคูณ `x * (tr - tl + 1)` ด้วย
:::

## ป้ายสองแบบ: ตั้งค่า กับ บวก

ทีนี้ถ้ามีคำสั่งสองแบบปนกัน คือ **"ตั้งทุกตัวใน $[l, r]$ ให้เป็น $x$"** (assign) กับ **"บวก $x$ ให้ทุกตัวใน $[l, r]$"** (add) ป้ายตัวเดียวไม่พอแล้ว และ **ลำดับของงานสำคัญมาก**: "ตั้งเป็น $5$ แล้วบวก $3$" ได้ $8$ แต่ "บวก $3$ แล้วตั้งเป็น $5$" ได้ $5$

ทริกคือบังคับให้ป้ายของทุกโหนดอยู่ในรูปเดียวเสมอ: **"ตั้งเป็น $S$ (ถ้ามี) แล้วค่อยบวก $A$"** แล้วดูว่างานใหม่ที่มาต่อท้ายรวมกับป้ายเดิมยังไง

| ป้ายที่ค้างอยู่ | งานใหม่ | ป้ายหลังรวม |
|---|---|---|
| บวก $A$ | บวก $B$ | บวก $A + B$ |
| บวก $A$ | ตั้งเป็น $T$ | ตั้งเป็น $T$ (บวกเก่าถูกทับ ทิ้งไปเลย) |
| ตั้งเป็น $S$ | บวก $B$ | ตั้งเป็น $S + B$ |
| ตั้งเป็น $S$ | ตั้งเป็น $T$ | ตั้งเป็น $T$ |

สังเกตว่าหลังรวมแล้ว ป้ายยังอยู่ในรูป "ตั้ง (ถ้ามี) แล้วบวก" เสมอ จริง ๆ แล้วรวมได้แค่ "ตั้งอย่างเดียว" หรือ "บวกอย่างเดียว" ด้วยซ้ำ ตอน `push` ก็ส่งตามลำดับเดียวกัน: ส่ง **ตั้งค่าก่อน แล้วค่อยส่งบวก**

เขียนกับโหนดที่เก็บผลรวมได้ประมาณนี้

```cpp
// ป้ายของโหนด v: has_set[v], set_val[v], add_val[v]  (ความหมาย: ตั้งเป็น set_val ถ้ามี แล้วบวก add_val)
void apply_set(int v, int len, long long x) {
  sum[v] = x * len;
  has_set[v] = true;
  set_val[v] = x;
  add_val[v] = 0;              // บวกที่ค้างอยู่ใต้การตั้งค่า ไม่มีความหมายแล้ว
}

void apply_add(int v, int len, long long x) {
  sum[v] += x * len;
  if (has_set[v]) set_val[v] += x;   // "ตั้งเป็น S แล้วบวก x" = "ตั้งเป็น S + x"
  else add_val[v] += x;
}
```

:::hint ใบ้ Greenhouse Rows
ใช้ตารางข้างบนได้ตรง ๆ แต่โหนดเก็บค่ามากสุดแทนผลรวม ลองถามตัวเองว่า ถ้าทุกตัวในช่วงถูกตั้งเป็น $x$ ค่ามากสุดเป็นเท่าไร และถ้าทุกตัวบวก $x$ ค่ามากสุดเปลี่ยนยังไง (ต้องคูณความยาวไหม?) แล้วเขียน `push` ให้ส่งตั้งค่าก่อนบวกเสมอ ระวัง: ตั้งค่าเป็น $0$ หรือเลขติดลบก็เป็นคำสั่งที่ถูกต้อง
:::

## จุดที่พลาดบ่อย

:::warning ลืม push ใน query
หลายคนใส่ `push` แค่ใน update เพราะคิดว่า query แค่อ่าน แต่ query ก็ลงไปอ่านลูก ซึ่งอาจยังเป็นค่าเก่า ทุกที่ที่จะลงไปหาลูกต้อง `push` ก่อน ไม่มีข้อยกเว้น
:::

:::warning ลืมคิดค่าของพ่อใหม่หลัง update
หลังลงไปแก้ลูกเสร็จต้อง `mn[v] = min(mn[2v], mn[2v+1])` (หรือ merge ของงานนั้น) เหมือนบทที่แล้ว ส่วน `apply` ไม่ต้องคิดจากลูก เพราะมันแก้ค่าของโหนดจากค่าเก่าของตัวเองได้เลย
:::

:::warning ใช้ค่า 0 แทน "ไม่มีป้ายตั้งค่า"
ถ้าเขียน `if (set_val[v] != 0)` คำสั่ง "ตั้งเป็น $0$" จะหายไปเฉย ๆ ให้ใช้ตัวแปร `bool` แยกต่างหากว่ามีป้ายตั้งค่าหรือเปล่า (ป้ายบวกใช้ $0$ เป็น "ไม่มี" ได้ เพราะบวก $0$ ไม่เปลี่ยนอะไร)
:::

:::warning ลำดับของป้ายสองแบบ
ตั้งค่าต้องล้างป้ายบวกเก่า, บวกที่มาทีหลังต้องไปรวมกับค่าที่ตั้ง และ `push` ต้องส่งตั้งค่าก่อนบวก สลับข้อไหนข้อหนึ่ง โปรแกรมจะยังตอบถูกกับตัวอย่างง่าย ๆ แต่ผิดเมื่อคำสั่งซ้อนกันหลายชั้น ทดสอบด้วยการเขียน brute force $O(NQ)$ แล้วสุ่มเทียบกับอาเรย์เล็ก ๆ หลายร้อยรอบ
:::

:::warning ป้ายคนละหน่วยกับค่าของโหนด
ในงานผลรวม ป้ายคือ "ตัวละ $x$" แต่ค่าของโหนดเพิ่ม $x \times$ ความยาว ถ้าเผลอส่งค่าที่คูณแล้วลงไปเป็นป้ายของลูก ค่าจะพองขึ้นทุกชั้น
:::

:::note ป้ายแบบอื่น ๆ
หลักเดียวกันใช้กับงานอื่นได้อีกมาก ขอแค่สองอย่าง: คำนวณค่าใหม่ของทั้งก้อนได้จากค่าเก่ากับป้าย และรวมป้ายสองอันที่มาต่อกันให้เป็นป้ายอันเดียวได้ เช่น **กลับบิตทั้งช่วง** ในอาเรย์ $0/1$ (จำนวนเลข $1$ กลายเป็นความยาวลบของเดิม และกลับสองครั้งเท่ากับไม่กลับ) หรือ **คูณแล้วบวก** $x \to bx + c$ (ซึ่งการตั้งค่าคือกรณี $b = 0$ และการบวกคือกรณี $b = 1$)

ถ้าเขียน Python: segment tree แบบ recursive ที่มี push จะช้าเกินไป ใช้แบบไม่ใช้ recursion ที่ push ลงตามเส้นทางของสองปลายก่อน แล้วค่อยทำงานแบบ bottom-up เหมือนบทที่แล้ว
:::

## สรุปสั้น ๆ

* Lazy propagation ทำให้ update ทั้งช่วงเป็น $O(\log N)$: โหนดที่อยู่ข้างในทั้งก้อนแก้ค่าตัวเองแล้วจดป้ายค้างไว้ ไม่ลงไปต่อ
* `apply(v, x)` แก้ค่าของโหนดและรวมงานเข้าป้าย, `push(v)` ส่งป้ายให้ลูกแล้วล้าง ต้อง `push` ทุกครั้งก่อนลงไปหาลูก ทั้งใน update และ query
* ค่าน้อยสุด/มากสุด: บวก $x$ ก็บวก $x$, ผลรวม: บวก $x \times$ ความยาว
* ป้ายสองแบบให้เก็บในรูป "ตั้ง (ถ้ามี) แล้วบวก" เสมอ ตั้งค่าล้างบวกเก่า บวกใหม่ไปรวมกับค่าที่ตั้ง push ตั้งค่าก่อนบวก
* ใช้ `bool` บอกว่ามีป้ายตั้งค่า อย่าใช้ $0$ และใช้ `long long` กับทุกอย่างที่สะสมได้
